import type {CertId,CertQuestion,CertTrack} from './certification-content';
// Original caselets. Each explores a decision, its verification and the demonstrated gap.
// These are related learning exercises, not three independent real-world incidents.
type Case={track:CertId;domain:string;scene:string;action:string;check:string;gap:string;id:string};
const cases:Case[]=[];
function group(track:CertId,text:string){for(const line of text.trim().split('\n')){const [domain,scene,action,check,gap]=line.split('|');if(!gap)throw Error('Incomplete authored case');cases.push({track,domain,scene,action,check,gap,id:`${track}-case-${cases.filter(c=>c.track===track).length+1}`});}}

export const practiceCaseByQuestion:Record<string,string>={};
export function makePracticeBank(tracks:CertTrack[],existing:readonly CertQuestion[]):CertQuestion[]{const out:CertQuestion[]=[];const prompts=['Which response most directly addresses the stated problem?','Which check most directly tests whether that problem was addressed?','Which gap is demonstrated by the evidence?'];for(const track of tracks){const own=cases.filter(c=>c.track===track.id),counts=new Map(track.domains.map(d=>[d.id,existing.filter(q=>q.track===track.id&&q.domain===d.id).length]));const candidates=own.flatMap((c,index)=>[0,1,2].map(facet=>{const field=(['action','check','gap'] as const)[facet],correct=c[field];const alternatives=own.filter(x=>x.id!==c.id).slice(index%Math.max(1,own.length-3)).concat(own).filter(x=>x.id!==c.id).filter((x,i,a)=>a.findIndex(y=>y[field]===x[field])===i).slice(0,3);const choices=[correct,...alternatives.map(x=>x[field])],explanations=[facet===2?`The supplied evidence demonstrates ${c.gap.toLowerCase()}.`:`This directly addresses ${c.gap.toLowerCase()}.`,...alternatives.map(x=>`This concerns ${x.gap.toLowerCase()}, rather than the problem demonstrated here.`)];const rotation=(index+facet)%4,order=[0,1,2,3].map(n=>(n+rotation)%4),domain=track.domains.find(d=>d.id===c.domain)!;const aws:Partial<Record<CertId,string>>={'cloud-practitioner':'cloud-practitioner-02','ai-practitioner':'ai-practitioner-01','solutions-architect':'solutions-architect-associate-03','developer-associate':'developer-associate-03'};const pdf:Partial<Record<CertId,number>>={'istqb-foundation':3345,'test-automation':3435,'ai-testing':9558,'generative-ai-testing':6295},slug=aws[track.id],source=slug?`https://docs.aws.amazon.com/aws-certification/latest/${slug}/${slug}-domain${track.domains.indexOf(domain)+1}.html`:pdf[track.id]?`https://istqb.org/?download_id=${pdf[track.id]}&sdm_process_download=1`:track.guide;const id=`${c.id}-${facet+1}`;practiceCaseByQuestion[id]=c.id;return {id,track:track.id,domain:c.domain,objective:`${domain.title} · caselet ${index+1} · ${['decision','verification','interpretation'][facet]}`,prompt:`${c.scene} ${prompts[facet]}`,choices:order.map(n=>choices[n]),answer:order.indexOf(0),explanations:order.map(n=>explanations[n]),source,reviewed:'2026-11-15'};}));while(existing.filter(q=>q.track===track.id).length+out.filter(q=>q.track===track.id).length<50){const available=[...candidates].sort((a,b)=>counts.get(a.domain)!-counts.get(b.domain)!);const next=available[0];if(!next)throw Error(`Insufficient authored questions for ${track.id}`);out.push(next);counts.set(next.domain,counts.get(next.domain)!+1);candidates.splice(candidates.indexOf(next),1);}}return out;}
